class Sort
{ 
    private static void Merge(int[] arr, int[] left, int[] right)
    {
        int leftint = 0, j = 0, k = 0; 
        while (leftint < left.Length && j < right.Length) 
        {
            if (left[leftint] <= right[j]) 
                arr[k++] = left[leftint++]; 
            else
                arr[k++] = right[j++]; 
        }

        while (leftint < left.Length) 
            arr[k++] = left[leftint++]; 

        while (j < right.Length) 
            arr[k++] = right[j++]; 
    }

    public static void MergeSort(int[] arr) 
    {
        if (arr.Length <= 1) 
            return;

        int mid = arr.Length / 2;  

        int[] left = new int[mid]; 
        int[] right = new int[arr.Length - mid];

        for (int leftint = 0; leftint < mid; leftint++) 
            left[leftint] = arr[leftint]; 

        for (int leftint = mid; leftint < arr.Length; leftint++) 
            right[leftint - mid] = arr[leftint]; 

        MergeSort(left); 
        MergeSort(right); 

        Merge(arr, left, right); 
    }
